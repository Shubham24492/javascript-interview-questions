Event loop:
The event loop continuously checks whether the call stack is empty and, when it is, moves
eligible callbacks from the task queues(Macro and Micro) into the call stack for execution. It
allows JavaScript to perform asynchronous operations without blocking the main thread.
Memoization: is the process of caching the output of the function calls so that subsequent calls
are faster.
Const memoize = fn => {
Const cache = new Map();
Return value => {
Const cachedResult = cache.get(value)
if(cachedResult !== undefined) return cachedResult
Const result = fun(value)
cache.set(value, result)
Return result;
}
}

Closures (cross questions with examples):
"A closure is a function together with its lexical environment. When a function is returned or
passed somewhere else, it can still access variables from the scope where it was created, even
after that outer function has finished executing. Closures are commonly used for private state,
data encapsulation, function factories, callbacks, and maintaining state.
Example:
- Private Counter
function createCounter() {
let count = 0;
return function () {
count++;
return count;
};
}
const counter = createCounter();
console.log(counter()); // 1
console.log(counter()); // 2
console.log(counter()); // 3
- One important use of closures is data privacy / encapsulation.
function createBankAccount() {
let balance = 1000;
return {
getBalance() {
return balance;
},
deposit(amount) {
balance += amount;
}
};
}
const account = createBankAccount();
console.log(account.getBalance()); // 1000
account.deposit(500);
console.log(account.getBalance()); // 1500
// You cannot directly access balance
console.log(account.balance); // undefined
