//useResize window size, resize the window => return height and width updated


import React, {useState, useEffect} from 'react';


const useResize = (a)=>{
const [size, setSize] = useState({height: window.innerHeight, width: window.innerWidth});

useEffect(()=>{
	const hadleResize = () => {setSize({height: window.innerHeight, width: window.innerWidth})}
	window.addEventListener('resize', hadleResize, true );
	()=>{ window.removeEventListener('resize') }
},[a])

return size;
}

export default useResize;
